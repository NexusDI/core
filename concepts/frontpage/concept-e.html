    meridian.ts
    import { Token, defineModule, provide } from '@nexusdi/core';

// 1. Write the interface.
interface IShieldGrid {
  raise(level: number): void;
}
// 2. Make a token for it.
const SHIELDS = new Token<IShieldGrid>('ShieldGrid');

// 3. Write the class.
class ShieldGrid implements IShieldGrid {
  raise(level: number) { /* ... */ }
}
interface IHelm {
  engage(): void;
}
const HELM = new Token<IHelm>('Helm');
class Helm implements IHelm {
  static deps = [SHIELDS] as const;
  constructor(private shields: IShieldGrid) {}
  engage() { this.shields.raise(3); }
}

// 4. Bind the token to the class, and export it.
const Tactical = defineModule({
  name: 'Tactical',
  providers: [provide(SHIELDS, { useClass: ShieldGrid })],
  exports: [SHIELDS],
});
export const Bridge = defineModule({
  name: 'Bridge',
  imports: [Tactical], // remove this line and Nexus.check fails
  providers: [provide(HELM, { useClass: Helm })],
});
export const Meridian = defineModule({ name: 'Meridian', imports: [Bridge] });
