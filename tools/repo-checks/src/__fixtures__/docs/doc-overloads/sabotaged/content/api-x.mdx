# `@nexusdi/x`

## `ping`

```ts twoslash
import { Beacon, ping } from '@nexusdi/x';
ping([new Beacon('Meridian')]);
// ---cut---
ping('Meridian');
```

## `hail`

```ts twoslash
import { hail } from '@nexusdi/x';

hail('Meridian');
```

relay.send('Helm');
// @errors: 2769
relay.send([1, 2]);
